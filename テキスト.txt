<script>
function autoRatio(incomeMan){
  if(incomeMan < 400) return 30;
  if(incomeMan < 700) return 35;
  return 40;
}

function calc(){
  const incomeMan = Number(document.getElementById('income').value || 0);
  const incomeYen = incomeMan * 10000;
  if(!incomeYen){
    document.getElementById('result').innerHTML = '年収を入力してください。';
    return;
  }

  let ratio = Number(document.getElementById('ratio').value);
  if(!ratio) ratio = autoRatio(incomeMan);

  const rateYear = Number(document.getElementById('rate').value || 0);
  const years = Number(document.getElementById('years').value || 0);
  const other = Number(document.getElementById('other').value || 0);

  if(rateYear <= 0 || years <= 0){
    document.getElementById('result').innerHTML = '金利と返済期間を確認してください。';
    return;
  }

  const r = rateYear / 100 / 12;
  const n = years * 12;
  const monthlyLimit = incomeYen * (ratio/100) / 12 - other;
  if(monthlyLimit <= 0){
    document.getElementById('result').innerHTML = '他のローンが多すぎて借入余力がありません。';
    return;
  }

  const pow = Math.pow(1 + r, n);
  const principal = monthlyLimit * (pow - 1) / (r * pow);

  const fmt = v => v.toLocaleString('ja-JP');

  document.getElementById('result').innerHTML =
    '想定返済負担率：' + ratio.toFixed(1) + '%<br>' +
    '月返済可能額：約 ' + fmt(Math.round(monthlyLimit)) + ' 円<br>' +
    '借入可能額：約 ' + fmt(Math.round(principal)) + ' 円<br>' +
    '<span class="small">※概算です。実際の審査結果とは異なります。</span>';
}
</script>
